// POST /api/outreach/enrich
// Request:  { text: string }     // newline-separated URLs, max 15 (server re-validates)
// Response: {
//   succeeded: number,
//   failed:    Array<{ url: string, reason: string }>,
//   partial:   Array<{ url: string, reason: string }>,
//   enriched:  Record<string, {
//     topGames: string[] | null;
//     genre: string | null;
//     email: string | null;
//     subscriberCount: number | null;
//     medianViews: number | null;
//   }>,                                // keyed by canonical URL (= row.url for selected rows)
// }
//
// IMPORTANT:
// - `succeeded` is a NUMBER (count), not an array. `failed`/`partial` are arrays.
// - `enriched[url]` is populated for both succeeded AND partial entries (the row reached
//   the upsert step). It is NOT populated for `failed` URLs.
// - For `partial` rows, `enriched[url].topGames` and `.genre` are null but the InnerTube-
//   derived fields (`email`, `subscriberCount`, `medianViews`) carry real values.
// (Verified against the Plan 01 Task 3 implementation.)
```

    async function handleSearch(e: React.FormEvent) {
      e.preventDefault();
      const q = keyword.trim();
      if (!q || searching) return;
      setSearching(true);
      setSearchError('');
      setSummary(null);
      setRowSelection({});
      try {
        const res = await fetch('/api/outreach/discover', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ keyword: q }),
        });
        const data = await res.json();
        if (!res.ok) {
          setSearchError(data.error ?? 'Search failed — please try again. If the problem persists, try a different keyword.');
          setRows([]);
        } else {
          const list = (data.channels as DiscoveredChannel[] | undefined) ?? [];
          setRows(list.map(c => ({ ...c, status: 'idle' as RowStatus })));
          setSearched(true);
        }
      } catch (err) {
        console.error('[discovery] search fetch failed', err);
        setSearchError('Search failed — please check your connection and try again.');
        setRows([]);
      } finally {
        setSearching(false);
      }
    }
    ```

    **handleSave (consumes the `enriched` map for in-place row fill-in — D-06):**
    ```typescript
    async function handleSave() {
      if (selectedCount === 0 || tooMany || saving) return;

      const selectedChannelIds = new Set(selectedRows.map(r => r.original.channelId));
      // Build url-by-channelId so we can match the response's `enriched` map (keyed by canonical URL).
      const urlByChannelId = new Map<string, string>(
        selectedRows.map(r => [r.original.channelId, r.original.url]),
      );
      const urls = selectedRows.map(r => r.original.url);

      // Mark all selected rows as 'saving' (D-06 step 1: spinner).
      setRows(prev => prev.map(r =>
        selectedChannelIds.has(r.channelId) ? { ...r, status: 'saving' as RowStatus } : r,
      ));
      setSaving(true);
      setSaveError('');
      setSummary(null);

      try {
        const res = await fetch('/api/outreach/enrich', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ text: urls.join('\n') }),
        });
        const data = await res.json();

        if (!res.ok) {
          setSaveError(data.error ?? 'Save failed — please try again.');
          // Revert statuses
          setRows(prev => prev.map(r =>
            selectedChannelIds.has(r.channelId) ? { ...r, status: 'idle' as RowStatus } : r,
          ));
          return;
        }

        // Build URL -> outcome maps from the enrich response.
        const failedUrls  = new Set<string>((data.failed  ?? []).map((x: { url: string }) => x.url));
        const partialUrls = new Set<string>((data.partial ?? []).map((x: { url: string }) => x.url));
        const enrichedMap: Record<string, EnrichedRow> = (data.enriched as Record<string, EnrichedRow> | undefined) ?? {};

        // D-06: update row statuses AND fill in enriched columns on the same row.
        // failed rows: status -> 'failed', keep prior cell values, NO enriched record.
        // partial rows: status -> 'partial', alreadySaved -> true, fill in InnerTube-
        //               derived fields from enriched[url] (topGames/genre will be null).
        // saved rows:   status -> 'saved', alreadySaved -> true, fill in ALL fields
        //               from enriched[url].
        setRows(prev => prev.map(r => {
          if (!selectedChannelIds.has(r.channelId)) return r;

          if (failedUrls.has(r.url)) {
            return { ...r, status: 'failed' as RowStatus };
          }

          const url = urlByChannelId.get(r.channelId) ?? r.url;
          const e = enrichedMap[url];

          const enrichedPatch = e
            ? {
                topGames:        e.topGames,
                genre:           e.genre,
                email:           e.email,
                // Promote real subscriber count from enrichment (was null at discover time).
                subscriberCount: e.subscriberCount ?? r.subscriberCount,
              }
            : {};

          if (partialUrls.has(r.url)) {
            return { ...r, ...enrichedPatch, status: 'partial' as RowStatus, alreadySaved: true };
          }
          return { ...r, ...enrichedPatch, status: 'saved' as RowStatus, alreadySaved: true };
        }));
        setSummary({
          succeeded: data.succeeded ?? 0,
          failed:    (data.failed   ?? []).length,
          partial:   (data.partial  ?? []).length,
        });
        setRowSelection({});
      } catch (err) {
        console.error('[discovery] save fetch failed', err);
        setSaveError('Save failed — your changes were not saved. Please try again.');
        setRows(prev => prev.map(r =>
          selectedChannelIds.has(r.channelId) ? { ...r, status: 'idle' as RowStatus } : r,
        ));
      } finally {
        setSaving(false);
      }
    }
    ```
