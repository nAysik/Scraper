const SOCIAL_SKIP = new Set([
  'youtube.com', 'www.youtube.com',
  'twitter.com', 'x.com',
  'instagram.com', 'www.instagram.com',
  'twitch.tv', 'www.twitch.tv',
  'tiktok.com', 'www.tiktok.com',
  'facebook.com', 'www.facebook.com',
]);

function unwrapYouTubeRedirect(raw: string): string {
  try {
    const u = new URL(raw.startsWith('http') ? raw : `https://www.youtube.com${raw}`);
    if (
      (u.hostname === 'www.youtube.com' || u.hostname === 'youtube.com') &&
      u.pathname === '/redirect'
    ) {
      return decodeURIComponent(u.searchParams.get('q') ?? '') || raw;
    }
    return raw;
  } catch {
    return raw;
  }
}
```

// Website email fallback (EML-01, EML-02, EML-03).
// Walks primary_links from the About page; skips social platforms; fetches the
// first qualifying URL with a 5-second timeout; runs EMAIL_RE on the HTML.
// All failures are silent — website fetch never blocks enrichment.
let websiteEmail: string | null = null;
try {
  const primaryLinks: any[] = (about as any)?.primary_links ?? [];
  for (const link of primaryLinks) {
    const raw: string = (link as any)?.endpoint?.metadata?.url ?? '';
    if (!raw) continue;
    const target = unwrapYouTubeRedirect(raw);
    if (!target) continue;
    let hostname = '';
    try { hostname = new URL(target).hostname; } catch { continue; }
    if (SOCIAL_SKIP.has(hostname)) continue;
    // Found a qualifying website URL — fetch it.
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 5000);
    try {
      const res = await fetch(target, {
        signal: controller.signal,
        headers: { 'User-Agent': 'Mozilla/5.0' },
      });
      const html = await res.text();
      const match = html.match(EMAIL_RE);
      if (match) websiteEmail = match[0];
    } finally {
      clearTimeout(timeoutId);
    }
    break; // only try the first qualifying link
  }
} catch {
  // silent — website fetch failure never blocks enrichment
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| server → external website | The enrichment server fetches an arbitrary URL from a channel's About page; the response is untrusted HTML from a third-party origin |
| YouTube About → primary_links | Link URLs are provided by YouTube's InnerTube API; redirect wrapper is trusted but the unwrapped target is not |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01 | Information Disclosure | unwrapYouTubeRedirect() | mitigate | Only the `q` param is used; full URL is re-parsed with `new URL()` before any fetch — malformed URLs are caught and skipped |
| T-06-02 | Spoofing | SOCIAL_SKIP hostname check | mitigate | Hostname is extracted via `new URL(target).hostname` after unwrapping — not a string-contains check — preventing bypasses like `youtube.com.evil.com` |
| T-06-03 | Denial of Service | fetch() against external site | mitigate | AbortController with 5-second timeout; single URL attempted per channel; failure path is fully caught and silent |
| T-06-04 | Information Disclosure | EMAIL_RE applied to fetched HTML | accept | Regex only extracts email-shaped strings already publicly present in the page HTML; no PII is introduced that wasn't already public |
| T-06-05 | Elevation of Privilege | Server-Side Request Forgery (SSRF) | mitigate | Only URLs originating from YouTube's own primary_links structure are fetched; SOCIAL_SKIP blocks same-origin YouTube redirects; no user-supplied URLs are fetched directly |
</threat_model>
