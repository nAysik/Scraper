```typescript
// src/app/api/outreach/find-emails/route.ts
// POST /api/outreach/find-emails
// Auth-gated batch email reveal sweep for existing outreach channels.
//
// Request:  { youtubeIds: string[] }  — IDs of channels to check (typically visible no-email rows)
// Response: { updated: number, failed: number }
//
// For each youtubeId:
//   1. getChannel(id) + getAbout() to read can_reveal_email
//   2. If true: call email_reveal endpoint, extract email via regex
//   3. If email found: UPDATE outreach_channels SET email = ? WHERE youtube_id = ?
// Failures per-channel are silent — they increment failed counter, never throw.

import { NextResponse, type NextRequest } from 'next/server';
import { createClient, createServiceClient } from '@/lib/supabase/server';
import { getClient } from '@/lib/scraper/innertube';

export const maxDuration = 300;

const EMAIL_RE = /[\w.+-]+@[\w-]+\.[a-z]{2,}/i;

export async function POST(request: NextRequest) {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  const body = await request.json().catch(() => ({}));
  const youtubeIds: string[] = Array.isArray(body.youtubeIds) ? body.youtubeIds : [];

  if (youtubeIds.length === 0) {
    return NextResponse.json({ updated: 0, failed: 0 });
  }

  let updated = 0;
  let failed = 0;

  try {
    const innertube = await getClient();
    const sb = createServiceClient();

    for (const youtubeId of youtubeIds) {
      try {
        const channel = await innertube.getChannel(youtubeId);
        const about = await channel.getAbout().catch(() => null);
        const canReveal = (about as any)?.can_reveal_email === true;

        if (!canReveal) continue;

        const revealEndpoint = (about as any)?.email_reveal;
        if (!revealEndpoint?.call) continue;

        const revealResult = await revealEndpoint.call((innertube as any).actions);
        const resultStr = JSON.stringify(revealResult ?? '');
        const emailMatch = resultStr.match(EMAIL_RE);
        if (!emailMatch) continue;

        const foundEmail = emailMatch[0];
        const { error } = await sb
          .from('outreach_channels')
          .update({ email: foundEmail })
          .eq('youtube_id', youtubeId)
          .eq('platform', 'youtube');

        if (error) {
          console.error('[find-emails] update failed', youtubeId, error.message);
          failed++;
        } else {
          updated++;
        }
      } catch (err) {
        console.error('[find-emails] channel error', youtubeId, err);
        failed++;
      }
    }

    return NextResponse.json({ updated, failed });
  } catch (err) {
    console.error('[find-emails] outer', err);
    return NextResponse.json({ error: 'Internal error' }, { status: 500 });
  }
}
```

Key points:
- Auth gate is first, matching all other outreach routes
- `maxDuration = 300` to handle large batches
- Per-channel errors increment `failed` and continue — never throw
- Only updates rows with `platform = 'youtube'` to avoid accidentally writing to Twitch rows
- Uses `createServiceClient()` for the UPDATE (bypass RLS), `createClient()` only for auth check

```typescript
async function handleFindEmails() {
  if (findingEmails) return;
  const noEmailRows = table.getRowModel().rows.filter(r =>
    !r.original.email && r.original.platform === 'youtube'
  );
  if (noEmailRows.length === 0) return;

  setFindingEmails(true);
  setFindProgress(`Checking ${noEmailRows.length} channels...`);

  try {
    const res = await fetch('/api/outreach/find-emails', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ youtubeIds: noEmailRows.map(r => r.original.youtubeId) }),
    });
    const data = await res.json();
    setFindProgress(`Found ${data.updated as number} new email${(data.updated as number) === 1 ? '' : 's'}`);

    // Re-fetch to show updated emails in table
    const refreshRes = await fetch('/api/outreach/channels');
    const refreshData = await refreshRes.json();
    const list: OutreachRow[] = (refreshData.channels ?? []).map((c: Omit<OutreachRow, 'status'>) => ({
      ...c,
      platform:      (c as OutreachRow).platform      ?? 'youtube',
      priorityScore:  (c as OutreachRow).priorityScore  ?? null,
      priorityReason: (c as OutreachRow).priorityReason ?? null,
      status: 'idle' as OutreachRowStatus,
    }));
    setRows(list);
  } catch {
    setFindProgress('Error - please try again.');
  } finally {
    setFindingEmails(false);
  }
}
```
