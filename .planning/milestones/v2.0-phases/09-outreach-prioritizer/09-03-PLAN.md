**2. Add handleScoreAll function** — insert after the `handleReenrichAll` function body, before the `handleDelete` declaration:
```typescript
async function handleScoreAll() {
  if (scoring || !gameName.trim()) return;
  setScoring(true);
  setScoreProgress('Scoring…');
  setToolbarError('');
  try {
    const res = await fetch('/api/outreach/score-all', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ gameName: gameName.trim(), comparables: comparables.trim() }),
    });
    if (!res.ok) {
      const data = await res.json() as { error?: string };
      setToolbarError(data.error ?? 'Scoring failed — please try again.');
      return;
    }
    // Re-fetch channels to pull in updated priority_score / priority_reason
    setScoreProgress('Refreshing…');
    const refreshRes = await fetch('/api/outreach/channels');
    const refreshData = await refreshRes.json() as { channels?: Omit<OutreachRow, 'status'>[] };
    if (refreshData.channels) {
      setRows(refreshData.channels.map((c) => ({
        ...c,
        platform:      c.platform ?? 'youtube',
        priorityScore: c.priorityScore ?? null,
        priorityReason: c.priorityReason ?? null,
        status:        'idle' as OutreachRowStatus,
      })));
    }
    setScoreForm(false);
    setScoreProgress('');
  } catch {
    setToolbarError('Scoring failed — please try again.');
  } finally {
    setScoring(false);
    if (scoreProgress !== '') setScoreProgress('');
  }
}
```

Note: the score-all route processes all batches server-side and returns only when done, so there is no per-batch progress from the server. The "Scoring batch N of M…" display is not achievable with a single blocking request — show "Scoring…" while the request is in flight. This matches the decisions (D-07 says "shows batch progress while running") at the level the API supports without streaming.
