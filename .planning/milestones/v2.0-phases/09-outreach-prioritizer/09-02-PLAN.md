```typescript
// src/lib/outreach/score-channels.ts
// Scores a batch of YouTube channels 1–10 for outreach fit using gpt-4o-mini.
// Uses strict JSON Schema (additionalProperties: false at every level, all props required).
// Called by POST /api/outreach/score-all.

import OpenAI from 'openai';

export interface ChannelToScore {
  youtubeId: string;
  name: string;
  genre: string | null;
  topGames: string[] | null;
}

export interface ScoreResult {
  youtubeId: string;
  score: number;   // integer 1–10
  reason: string;  // one sentence
}

let openai: OpenAI | null = null;
function getOpenAI(): OpenAI {
  if (!openai) openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY! });
  return openai;
}

const BATCH_SCHEMA = {
  type: 'object',
  additionalProperties: false,
  properties: {
    scores: {
      type: 'array',
      items: {
        type: 'object',
        additionalProperties: false,
        properties: {
          youtubeId: { type: 'string' },
          score:     { type: 'number' },
          reason:    { type: 'string' },
        },
        required: ['youtubeId', 'score', 'reason'],
      },
    },
  },
  required: ['scores'],
} as const;

const OPENAI_TIMEOUT_MS = 30_000;

export async function scoreBatch(
  channels: ChannelToScore[],
  gameName: string,
  comparables: string,
): Promise<ScoreResult[]> {
  const client = getOpenAI();

  const systemPrompt =
    `You score YouTube gaming channels 1–10 for outreach fit for "${gameName}", ` +
    `a game similar to ${comparables}. ` +
    `10 = perfect fit (same genre, directly comparable audience). ` +
    `1 = completely unrelated. ` +
    `For each channel, return its youtubeId, an integer score 1–10, and a one-sentence reason. Be concise.`;

  const userContent = JSON.stringify(
    channels.map(c => ({
      youtubeId: c.youtubeId,
      name:      c.name,
      genre:     c.genre ?? 'unknown',
      topGames:  c.topGames ?? [],
    })),
  );

  const completionPromise = client.chat.completions.parse({
    model: 'gpt-4o-mini',
    temperature: 0,
    messages: [
      { role: 'system', content: systemPrompt },
      { role: 'user',   content: userContent },
    ],
    response_format: {
      type: 'json_schema',
      json_schema: {
        name: 'channel_scores',
        strict: true,
        schema: BATCH_SCHEMA,
      },
    },
  });

  const completion = await Promise.race([
    completionPromise,
    new Promise<never>((_, reject) =>
      setTimeout(() => reject(new Error('openai_timeout')), OPENAI_TIMEOUT_MS),
    ),
  ]);

  const parsed = completion.choices[0]?.message?.parsed as { scores: ScoreResult[] } | null;
  if (!parsed) throw new Error('No parsed result from gpt-4o-mini');

  // Clamp scores to 1–10 and coerce to integer defensively
  return parsed.scores.map(s => ({
    youtubeId: s.youtubeId,
    score:     Math.min(10, Math.max(1, Math.round(s.score))),
    reason:    s.reason,
  }));
}
```

Create the directory src/app/api/outreach/score-all/ and write route.ts. The route:

1. Auth-gates with the standard supabase.auth.getUser() pattern.
2. Parses { gameName, comparables } from the JSON body — returns 400 if gameName is empty after trim.
3. Fetches all outreach_channels rows where platform = 'youtube' OR platform IS NULL, selecting youtube_id, name, genre, top_games. Uses the auth client (not service role) for the read — the session user has read access via RLS.
4. Chunks the fetched channels into batches of 20.
5. For each batch: calls scoreBatch(), then for each ScoreResult upserts priority_score and priority_reason via service role update (eq youtube_id). Individual update failures are counted in failed, not fatal.
6. Returns { scored: number, failed: number } when all batches complete.

```typescript
// src/app/api/outreach/score-all/route.ts
// POST /api/outreach/score-all { gameName: string, comparables: string }
// Auth-gated. Scores all YouTube channels in batches of 20 via gpt-4o-mini.
// Processes batches sequentially. Returns { scored, failed } on completion.

import { NextRequest, NextResponse } from 'next/server';
import { createClient, createServiceClient } from '@/lib/supabase/server';
import { scoreBatch, type ChannelToScore } from '@/lib/outreach/score-channels';

export const maxDuration = 300;

const BATCH_SIZE = 20;

export async function POST(request: NextRequest) {
  // Auth check
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  // Parse and validate body
  let gameName: string;
  let comparables: string;
  try {
    const body = await request.json() as { gameName?: unknown; comparables?: unknown };
    if (typeof body.gameName !== 'string' || !body.gameName.trim()) {
      return NextResponse.json({ error: 'gameName is required' }, { status: 400 });
    }
    gameName    = body.gameName.trim().slice(0, 200);
    comparables = typeof body.comparables === 'string' ? body.comparables.trim().slice(0, 500) : '';
  } catch {
    return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 });
  }

  // Fetch all YouTube channels (platform = 'youtube' or NULL legacy rows)
  const { data: channelRows, error: fetchError } = await supabase
    .from('outreach_channels')
    .select('youtube_id, name, genre, top_games, platform')
    .or('platform.eq.youtube,platform.is.null');

  if (fetchError) {
    return NextResponse.json({ error: fetchError.message }, { status: 500 });
  }

  const channels: ChannelToScore[] = (channelRows ?? []).map((c: Record<string, unknown>) => ({
    youtubeId: c.youtube_id as string,
    name:      c.name      as string,
    genre:     (c.genre    as string | null) ?? null,
    topGames:  (c.top_games as string[] | null) ?? null,
  }));

  if (channels.length === 0) {
    return NextResponse.json({ scored: 0, failed: 0 });
  }

  // Process batches sequentially
  const sb = createServiceClient();
  let scored = 0;
  let failed = 0;

  for (let i = 0; i < channels.length; i += BATCH_SIZE) {
    const batch = channels.slice(i, i + BATCH_SIZE);

    let results;
    try {
      results = await scoreBatch(batch, gameName, comparables);
    } catch {
      // Entire batch failed (GPT call failed) — count all as failed
      failed += batch.length;
      continue;
    }

    // Persist each score result individually; individual DB failures don't abort the batch
    const resultMap = new Map(results.map(r => [r.youtubeId, r]));
    for (const ch of batch) {
      const result = resultMap.get(ch.youtubeId);
      if (!result) {
        // GPT didn't return a score for this channel
        failed++;
        continue;
      }
      const { error: updateError } = await sb
        .from('outreach_channels')
        .update({ priority_score: result.score, priority_reason: result.reason })
        .eq('youtube_id', ch.youtubeId);
      if (updateError) {
        failed++;
      } else {
        scored++;
      }
    }
  }

  return NextResponse.json({ scored, failed });
}
```
