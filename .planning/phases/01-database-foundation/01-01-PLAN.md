```typescript
// scripts/verify-outreach-channels.ts
// One-shot smoke test for the outreach_channels table.
// Run with: npx tsx scripts/verify-outreach-channels.ts
// Requires .env.local with NEXT_PUBLIC_SUPABASE_URL, NEXT_PUBLIC_SUPABASE_ANON_KEY, SUPABASE_SERVICE_ROLE_KEY.

import { createClient } from '@supabase/supabase-js';
import { config } from 'dotenv';
import { resolve } from 'path';

config({ path: resolve(process.cwd(), '.env.local') });

const url = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const anonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!;
const serviceKey = process.env.SUPABASE_SERVICE_ROLE_KEY!;

if (!url || !anonKey || !serviceKey) {
  console.error('Missing required env vars in .env.local');
  process.exit(1);
}

const service = createClient(url, serviceKey);
const anon = createClient(url, anonKey);

const TEST_YOUTUBE_ID = `__verify_${Date.now()}`;
const TEST_URL = `https://youtube.com/__verify_${Date.now()}`;

let exitCode = 0;
function fail(msg: string, err?: unknown) {
  console.error(`FAIL: ${msg}`, err ?? '');
  exitCode = 1;
}
function ok(msg: string) {
  console.log(`OK: ${msg}`);
}

// 1. Service role INSERT
const { data: inserted, error: insErr } = await service
  .from('outreach_channels')
  .insert({
    youtube_id: TEST_YOUTUBE_ID,
    name: 'Verify Channel',
    url: TEST_URL,
    subscriber_count: 1234,
    top_games: ['Minecraft', 'Stardew Valley'],
    genre: 'Cozy',
    median_views: 5000,
    last_enriched_at: new Date().toISOString(),
  })
  .select()
  .single();

if (insErr || !inserted) {
  fail('service role insert', insErr);
} else {
  ok(`service role insert (id=${inserted.id})`);
}

// 2. Anon SELECT must be rejected (or return empty due to RLS)
const { data: anonRows, error: anonErr } = await anon
  .from('outreach_channels')
  .select('*')
  .eq('youtube_id', TEST_YOUTUBE_ID);

// RLS for SELECT typically returns an empty result set for anon (not an error).
// The policy `using (auth.role() = 'authenticated')` filters anon rows out.
if (anonErr) {
  ok(`anon select rejected with error (acceptable): ${anonErr.message}`);
} else if (!anonRows || anonRows.length === 0) {
  ok('anon select returned 0 rows (RLS filtered as expected)');
} else {
  fail(`anon select returned ${anonRows.length} rows — RLS is NOT blocking anon`);
}

// 3. Schema shape via information_schema (service role)
const { data: cols, error: colErr } = await service.rpc('exec_sql', {
  sql: `select column_name, data_type, is_nullable from information_schema.columns where table_name = 'outreach_channels' order by ordinal_position;`,
}).select?.() ?? { data: null, error: null };

// Fallback: query directly via service client when exec_sql RPC is not present.
const { data: serviceRows, error: serviceReadErr } = await service
  .from('outreach_channels')
  .select('*')
  .eq('youtube_id', TEST_YOUTUBE_ID)
  .single();

if (serviceReadErr || !serviceRows) {
  fail('service role read-back', serviceReadErr);
} else {
  const expectedKeys = [
    'id', 'youtube_id', 'name', 'url', 'subscriber_count',
    'top_games', 'genre', 'median_views', 'last_enriched_at', 'created_at',
  ];
  const missing = expectedKeys.filter((k) => !(k in serviceRows));
  if (missing.length) fail(`schema missing columns: ${missing.join(', ')}`);
  else ok('schema has all 10 expected columns');

  if (!Array.isArray(serviceRows.top_games)) fail('top_games is not an array');
  else ok('top_games is an array');
}

// Cleanup
const { error: delErr } = await service
  .from('outreach_channels')
  .delete()
  .eq('youtube_id', TEST_YOUTUBE_ID);
if (delErr) fail('cleanup delete', delErr);
else ok('cleanup delete');

process.exit(exitCode);
```

|----------|-------------|

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information Disclosure | `outreach_channels` SELECT via anon key | mitigate | RLS enabled with `using (auth.role() = 'authenticated')`. Anon role is NOT 'authenticated', so anon reads return 0 rows. Verified end-to-end in Task 3. |
| T-01-02 | Tampering | `outreach_channels` INSERT/UPDATE/DELETE via anon key | mitigate | RLS enabled with no policy granting INSERT/UPDATE/DELETE to authenticated or anon. By Postgres RLS semantics, with RLS on and no permissive write policy, all writes from non-bypass roles are denied. Service role (used by routes) bypasses RLS by design. |
| T-01-03 | Elevation of Privilege | Overly permissive RLS policy (e.g. `using (true)` or `for all`) | mitigate | Policy is scoped `for select` only and uses `auth.role() = 'authenticated'`. No `for all`, no `using (true)`. Verified by acceptance criterion that file does not contain `for insert` or `for update` policy clauses. |
| T-01-04 | Information Disclosure | `SUPABASE_SERVICE_ROLE_KEY` leakage to the browser | mitigate | Key lives only in `.env.local` (server-only) and is read in `src/lib/supabase/server.ts:createServiceClient()`. The client is invoked only from server contexts (Route Handlers, cron). No code path imports `createServiceClient` into a `'use client'` file. (No new code paths added in this phase — verified by no source file changes outside the migration SQL and the standalone verification script.) |
| T-01-05 | Tampering | SQL injection in migration DDL | accept | DDL is fully literal, statically authored, and contains no user input. There is nothing to inject into. |
| T-01-06 | Repudiation | No audit trail of who applied the migration | accept | The migration is applied once by the developer via the Supabase Dashboard, which logs SQL executions in the project's audit log. Application-level audit is out of scope for v1 (and out of scope per REQUIREMENTS.md). |
| T-01-07 | Denial of Service | Unbounded `top_games` array growth | accept | `top_games` is enforced by application code (Phase 2) to a max of 3 entries. Schema-level CHECK constraint is unnecessary at expected scale (low thousands of rows). |
| T-01-08 | Information Disclosure | RLS bypass via insecure custom database functions | mitigate | No SECURITY DEFINER functions, no triggers, no views are created in this migration. Verified by acceptance criterion that file content matches the literal SQL block (no `create function`, no `create trigger`, no `create view`). |
